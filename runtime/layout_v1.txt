# Sprout68k 共有ランタイム 第1版のメモリ配置（2026-08-22 凍結）
#
# ここが正典。シェルからは source で、生成器からは1行ずつ読んで使う。
# **借りた定数には見えない前提が付く**ので、成立していなければならない
# 不等式を tools/verify_runtime.mts が実測値と突き合わせて検査する
# （「たぶん入るだろう」で置かない）。
#
# 前提: 1MB機（RAM $000000〜$0FFFFF）。実機互換の要件として最小構成で成立させる。
ABI_VERSION=1
# ランタイム本体（.text/.rodata/.data）。先頭$3000はブートセクタが飛んでくる番地。
RUNTIME_BASE=0x003000
# ジャンプテーブルの先頭。$3000=BRA.W、$3004=ABI版、$3006=予約 の次。
# **この番地は利用者コードに焼き込まれる。未来永劫動かせない。**
ABI_TABLE_BASE=0x003008
# 利用者ペイロード（12バイトのヘッダ＋本体）の先頭。ここも焼き込まれる。
USER_BASE=0x010000
# 利用者が使える領域の終端（この番地は含まない）。ここから先はランタイムの裏バッファ。
USER_LIMIT=0x060000
# ランタイムの .bss（512KBの裏バッファを含む）の先頭。
RUNTIME_BSS_BASE=0x060000
# スタック（下方向に伸びる）。裏バッファの末尾との間にこのマージンを空ける。
STACK_ADDR=0x0F0000
STACK_MARGIN=0x001000
RAM_SIZE=0x100000
